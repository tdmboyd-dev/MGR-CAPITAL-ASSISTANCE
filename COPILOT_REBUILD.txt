import nodemailer from "nodemailer";

export const notificationService = {
sendEmail: async ({ to, subject, body }) => {
const transporter = nodemailer.createTransport({
host: process.env.SMTP_HOST,
port: Number(process.env.SMTP_PORT),
auth: {
user: process.env.SMTP_USER,
pass: process.env.SMTP_PASS
}
});

await transporter.sendMail({
from: process.env.SMTP_FROM,
to,
subject,
html: body
});

return { success: true };
}
};

import PDFDocument from "pdfkit";
import fs from "fs";

export const pdfService = {
generateAgreement: async (caseId, data) => {
const doc = new PDFDocument();
const filePath = `backend/storage/documents/${caseId}/agreement.pdf`;
fs.mkdirSync(`backend/storage/documents/${caseId}`, { recursive: true });
doc.pipe(fs.createWriteStream(filePath));
doc.text("SERVICE AGREEMENT");
doc.text(JSON.stringify(data, null, 2));
doc.end();
return filePath;
}
};

import fs from "fs";

export const documentVaultService = {
saveFile: async (caseId, file) => {
const dir = `backend/storage/documents/${caseId}`;
fs.mkdirSync(dir, { recursive: true });
const filePath = `${dir}/${file.originalname}`;
fs.writeFileSync(filePath, file.buffer);
return filePath;
}
};
